select titles.title_id, titles.title_name, authors.au_lname
from authors
join title_authors on title_authors.au_id = authors.au_id
join titles on titles.title_id = title_authors.title_id
where au_order = 1
;

Query 4:

select titles.type, count(*) as "Number"
from titles
join publishers on publishers.pub_id = titles.pub_id
where publishers.state = 'CA'
group by type
order by type
;

Query 10:

select hier.emp_id, employees.emp_name, hier.boss_id, imp.emp_name
from hier
left join employees on hier.emp_id = employees.emp_id
left join employees imp on hier.boss_id = imp.emp_id
order by hier.emp_id
;
