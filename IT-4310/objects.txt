CREATE TYPE address_ty AS OBJECT
(name VARCHAR2(30),
address VARCHAR2(30),
city VARCHAR2(30),
state CHAR(2),
zip CHAR(5));
/

DROP TABLE IF EXISTS professor;
CREATE TABLE professor (
    pid CHAR(5),
    address address_ty,
    salary VARCHAR2(30)
);
